{
  "mcpServers": {

        "@modelcontextprotocol/server-filesystem",
        "C:\\Users\\ahnda\\Downloads"
      ]
    },
    "youtube-transcript": {
      "command": "npx",
      "args": ["-y", "@kimtaeyoon83/mcp-server-youtube-transcript"]
    }
  }
}
